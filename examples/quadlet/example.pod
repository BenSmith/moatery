# The pod holds the network namespace: its infra container creates it,
# and it outlives any restart of the workload. The rules are loaded
# before the unit is started, so the listener units, ordered after it,
# bind into a namespace whose traffic is already redirected.
#
# keep-id maps this user to the same uid inside; every uid the pod's
# user namespace maps is served by the inspector.
#
# No pod cgroup: with one, every container's start asks systemd for the
# pod's slice unless podman finds its directory at the path it assumes,
# and a slice systemd already has fails that start.
#
# The image's hosts file, not one seeded from the host's: a name in it
# is never asked of the responder, and is dialled at the host's address
# for it.

[Unit]
Wants=moat-inspect-example.service moat-resolve-example.service

[Pod]
PodName=example
Network=pasta
UserNS=keep-id
ExitPolicy=continue
PodmanArgs=--hosts-file=image --share-parent=false

[Service]
ExecStartPost=%h/.local/bin/moat-pod-netns rules example
